```{r, eval = FALSE}
library(dplyr)
library(purrr)

# Get leaderboard and model IDs
lb <- as.data.frame(h2o.get_leaderboard(aml, extra_columns = "ALL"))
model_ids <- lb$model_id

# Extract model type and hyperparameters
hp_summary <- map_dfr(model_ids, function(id) {
  m <- h2o.getModel(id)
  
  # Extract model family (prefix of ID)
  family <- sub("_.*", "", id)
  
  # Extract parameter list
  params <- m@parameters
  
  # Keep only meaningful tuning parameters (exclude frame refs, seed, etc.)
  params <- params[!names(params) %in% c(
    "training_frame", "validation_frame", "nfolds", "seed", "model_id",
    "response_column", "ignored_columns", "fold_assignment", "fold_column"
  )]
  
  # Convert list to data frame (1 row per parameter)
  tibble(
    model_id = id,
    model_family = family,
    param_name = names(params),
    param_value = sapply(params, function(x) paste0(collapse = ", ", x))
  )
})

# Show summary
hp_summary %>%
  arrange(model_family, model_id, param_name) %>%
  print(n = 20)

# Keep only parameters that change across models
hp_variable <- hp_summary %>%
  group_by(param_name) %>%
  filter(n_distinct(param_value) > 1) %>%
  ungroup()

  arrange(model_family, model_id, param_name) %>%
```

```{r, eval = FALSE}
# ============================================================
# ⚙️ Load packages
# ============================================================
if (!requireNamespace("caret", quietly = TRUE)) install.packages("caret")
if (!requireNamespace("pROC", quietly = TRUE)) install.packages("pROC")
if (!requireNamespace("PRROC", quietly = TRUE)) install.packages("PRROC")

library(caret)
library(pROC)
library(PRROC)

set.seed(1234)

n <- 1000
p <- 20

# True signal only in first 5 predictors, rest are noise
X <- matrix(rnorm(n * p), n, p)
colnames(X) <- paste0("x", 1:p)

# Nonlinear, overlapping signal
lin_pred <- 0.5 * X[,1] - 0.3 * X[,2]^2 + 0.6 * sin(X[,3]) - 0.4 * X[,4] + 0.2 * X[,5]
prob <- 1 / (1 + exp(-lin_pred))
y <- rbinom(n, 1, prob)

df <- data.frame(X, y = factor(y, labels = c("no", "yes")))

# Train/test split (80/20)
set.seed(42)
train_index <- createDataPartition(df$y, p = 0.8, list = FALSE)
train_data <- df[train_index, ]
test_data  <- df[-train_index, ]

# ============================================================
# 🤖 2. Define training control with 5-fold CV + AUC
# ============================================================
ctrl <- trainControl(
  method = "cv",
  number = 5,
  classProbs = TRUE,
  summaryFunction = twoClassSummary,  # computes AUC, Sensitivity, Specificity
  savePredictions = "final"
)

# ============================================================
# 🚀 3. Train a model (e.g., logistic regression)
# ============================================================
set.seed(42)
model <- train(
  y ~ .,
  data = train_data,
  method = "glmnet",
  metric = "ROC",
  trControl = ctrl,
  tuneLength = 10
)

print(model)

# ============================================================
# 📊 4. Compute cross-validated AUC and AUPRC manually
# ============================================================
cv_preds <- model$pred
cv_auc <- pROC::roc(cv_preds$obs, cv_preds$yes, levels = rev(levels(cv_preds$obs)))$auc

# Use PRROC for AUPRC
cv_pr <- PRROC::pr.curve(
  scores.class0 = cv_preds$yes[cv_preds$obs == "yes"],
  scores.class1 = cv_preds$yes[cv_preds$obs == "no"],
  curve = TRUE
)
cv_auprc <- cv_pr$auc.integral

cat("\n=== Cross-validated performance ===\n")
cat("CV AUC:   ", round(cv_auc, 3), "\n")
cat("CV AUPRC: ", round(cv_auprc, 3), "\n")

# ============================================================
# 🧠 5. Evaluate on independent test set
# ============================================================
test_pred <- predict(model, newdata = test_data, type = "prob")

# Compute AUC
test_auc <- pROC::roc(test_data$y, test_pred$yes, levels = rev(levels(test_data$y)))$auc

# Compute AUPRC
test_pr <- PRROC::pr.curve(
  scores.class0 = test_pred$yes[test_data$y == "yes"],
  scores.class1 = test_pred$yes[test_data$y == "no"],
  curve = TRUE
)
test_auprc <- test_pr$auc.integral

cat("\n=== Independent test set performance ===\n")
cat("Test AUC:   ", round(test_auc, 3), "\n")
cat("Test AUPRC: ", round(test_auprc, 3), "\n")

# ============================================================
# 📈 6. (Optional) Plot PR curve on test set
# ============================================================
plot(
  test_pr$curve[, 1], test_pr$curve[, 2], type = "l", col = "red", lwd = 2,
  xlab = "Recall", ylab = "Precision",
  main = paste0("Precision-Recall Curve (AUPRC = ", round(test_auprc, 3), ")")
)
abline(h = mean(test_data$y == "yes"), lty = 2, col = "gray")
```

H2o wrapper function
```{r, eval = FALSE}
#' Run H2O AutoML and Return Cross-Validation Metrics
#'
#' This function runs H2O AutoML on the provided dataset and
#' returns CV-based performance metrics (AUC, AUPRC, Accuracy, LogLoss)
#' for all models in the leaderboard.
#'
#' @param train_df Data frame with predictors and target variable.
#' @param outcome_col Character string naming the binary target variable.
#' @param nfolds Number of CV folds (default = 5).
#' @param max_runtime_secs Max runtime in seconds for AutoML (default = 120).
#' @param seed Random seed (default = 1234).
#'
#' @return A tibble with model IDs, family, AUC, AUPRC, Accuracy, LogLoss (CV mean & SD).
#' @export
#'
run_h2o_benchmark <- function(train_df, outcome_col,
                              nfolds = 5,
                              max_runtime_secs = 120,
                              seed = 1234) {
  requireNamespace("h2o", quietly = TRUE)
  requireNamespace("dplyr", quietly = TRUE)

  # Initialize H2O if not running
  if (!h2o::h2o.clusterIsUp()) {
    h2o::h2o.init(nthreads = -1, max_mem_size = "8G")
  }

  # Prepare H2O frame
  df_h2o <- h2o::as.h2o(train_df)
  y <- outcome_col
  x <- setdiff(names(train_df), y)
  df_h2o[, y] <- as.factor(df_h2o[, y])

  message("🚀 Running H2O AutoML benchmark...")

  aml <- h2o::h2o.automl(
    x = x,
    y = y,
    training_frame = df_h2o,
    nfolds = nfolds,
    max_runtime_secs = max_runtime_secs,
    seed = seed,
    keep_cross_validation_predictions = TRUE
  )

  lb <- as.data.frame(h2o::h2o.get_leaderboard(aml, extra_columns = "ALL"))
  model_ids <- lb$model_id

  message("✅ Extracting cross-validation metrics from models...")

  results <- lapply(model_ids, function(id) {
    model <- h2o::h2o.getModel(id)
    perf <- model@model$cross_validation_metrics_summary

    auc_mean <- as.numeric(perf["auc", "mean"])
    auc_sd   <- as.numeric(perf["auc", "sd"])
    auprc_mean <- as.numeric(perf["pr_auc", "mean"])
    auprc_sd   <- as.numeric(perf["pr_auc", "sd"])
    acc_mean <- as.numeric(perf["accuracy", "mean"])
    acc_sd   <- as.numeric(perf["accuracy", "sd"])
    logloss_mean <- as.numeric(perf["logloss", "mean"])
    logloss_sd   <- as.numeric(perf["logloss", "sd"])

    model_family <- model@algorithm

    data.frame(
      model_id = id,
      model_family = model_family,
      AUC_mean = auc_mean,  AUC_SD = auc_sd,
      AUPRC_mean = auprc_mean, AUPRC_SD = auprc_sd,
      Accuracy_mean = acc_mean, Accuracy_SD = acc_sd,
      LogLoss_mean = logloss_mean, LogLoss_SD = logloss_sd,
      stringsAsFactors = FALSE
    )
  })

  results_df <- dplyr::bind_rows(results) %>%
    dplyr::arrange(dplyr::desc(AUC_mean))

  message("🏁 Benchmark complete! Returning summary table.")

  return(results_df)
}
```

```{r}
  # Safe init

    h2o::h2o.init(nthreads = -1)
  })

  # Convert to H2O frames
  train_h2o <- as.h2o(train_df)
  test_h2o  <- as.h2o(test_df)

  x <- setdiff(names(train_df), outcome_col)
  y <- outcome_col
  train_h2o[, y] <- as.factor(train_h2o[, y])
  test_h2o[, y]  <- as.factor(test_h2o[, y])

  # -------------------------------
  # AutoML for the INNER CV
  # -------------------------------
  aml <- h2o.automl(
    x = x,
    y = y,
    training_frame = train_h2o,
    nfolds = nfolds_inner,
    max_runtime_secs = max_runtime_secs,
    balance_classes = balance_classes,
    seed = seed,
    keep_cross_validation_predictions = TRUE,
    keep_cross_validation_models = TRUE
  )

  # -------------------------------
  # Extract ALL MODEL CV METRICS
  # -------------------------------
  lb <- h2o.get_leaderboard(aml, extra_columns = "ALL")

  model_ids <- as.data.frame(lb$model_id)[,1]

  all_cv_metrics <- lapply(model_ids, function(mid) {
    m <- h2o.getModel(mid)
    perf <- m@model$cross_validation_metrics_summary

    data.frame(
      model_id = mid,
      AUC      = as.numeric(perf["auc", "mean"]),
      AUPRC    = as.numeric(perf["pr_auc", "mean"]),
      logloss  = as.numeric(perf["logloss", "mean"])
    )
  })

  all_cv_metrics <- do.call(rbind, all_cv_metrics)

  # -------------------------------
  # Evaluate OUTER TEST FOLD
  # -------------------------------
  best_model <- aml@leader
  perf_outer <- h2o.performance(best_model, newdata = test_h2o)

  outer_auc   <- as.numeric(h2o.auc(perf_outer))
  outer_auprc <- as.numeric(h2o.pr_auc(perf_outer))

  list(
    inner_cv_metrics = all_cv_metrics,    # AUC across all models
    best_model_id    = best_model@model_id,
    outer_auc        = outer_auc,
    outer_auprc      = outer_auprc
  )
}
```

```{r}
run_h2o_fold <- function(train_df, test_df, outcome_col, 
                         max_runtime_secs = 60, seed = 1234,
                         nfolds_inner = 5, balance_classes = TRUE) {

  # Ensure H2O is running
  tryCatch({ h2o::h2o.init(nthreads = -1) }, 
           error = function(e) h2o::h2o.init(nthreads = -1))

  # Convert to H2O
  train_h2o <- as.h2o(train_df)
  test_h2o  <- as.h2o(test_df)

  x <- setdiff(names(train_df), outcome_col)
  y <- outcome_col
  train_h2o[, y] <- as.factor(train_h2o[, y])
  test_h2o[, y]  <- as.factor(test_h2o[, y])

  # ========================================
  # Run AutoML
  # ========================================
  aml <- h2o.automl(
    x = x,
    y = y,
    training_frame = train_h2o,
    nfolds = nfolds_inner,
    max_runtime_secs = max_runtime_secs,
    balance_classes = balance_classes,
    seed = seed,
    keep_cross_validation_predictions = TRUE,
    keep_cross_validation_models = TRUE
  )

  # ========================================
  # Extract leaderboard
  # ========================================
  lb <- h2o.get_leaderboard(aml, extra_columns = "ALL")

  model_ids <- as.data.frame(lb$model_id)[,1]

  # Extract family using model_id string
  model_families <- sub("_.*", "", model_ids)

  # ========================================
  # Extract CV metrics for ALL models
  # ========================================
  all_cv_metrics <- lapply(seq_along(model_ids), function(i) {
    mid <- model_ids[i]
    fam <- model_families[i]

    m <- h2o.getModel(mid)
    perf <- m@model$cross_validation_metrics_summary

    data.frame(
      model_id     = mid,
      model_family = fam,
      AUC          = as.numeric(perf["auc", "mean"]),
      AUPRC        = as.numeric(perf["pr_auc", "mean"]),
      logloss      = as.numeric(perf["logloss", "mean"])
    )
  })

  all_cv_metrics <- do.call(rbind, all_cv_metrics)

  # ========================================
  # Evaluate OUTER fold performance 
  # ========================================
  best_model <- aml@leader
  perf_outer <- h2o.performance(best_model, newdata = test_h2o)

  outer_auc   <- as.numeric(h2o.auc(perf_outer))
  outer_auprc <- as.numeric(h2o.pr_auc(perf_outer))

  list(
    inner_cv_metrics = all_cv_metrics,
    best_model_id    = best_model@model_id,
    outer_auc        = outer_auc,
    outer_auprc      = outer_auprc
  )
}
```

```{r}
library(caret)
library(dplyr)
library(ggplot2)
library(mlbench)
library(h2o)

data("Sonar")
df <- Sonar
outcome_col <- "Class"

set.seed(42)
folds <- caret::createMultiFolds(df[[outcome_col]], k = 3, times = 1)

all_results <- list()
plot_list <- list()
counter <- 1

for (fold_name in names(folds)) {

  train_idx <- folds[[fold_name]]
  test_idx  <- setdiff(seq_len(nrow(df)), train_idx)

  train_df <- df[train_idx, ]
  test_df  <- df[test_idx, ]

  # Remove near-zero variance (train only)
  nzv <- caret::nearZeroVar(train_df[, -ncol(train_df)])
  if (length(nzv) > 0) {
    train_df <- train_df[, -nzv, drop = FALSE]
    test_df <- test_df[, names(train_df), drop = FALSE]
  }

  cat("Running outer fold:", fold_name, "\n")

  res <- run_h2o_fold(train_df, test_df, outcome_col, nfolds_inner = 5,
                      max_runtime_secs = 20)

  # Store results
  all_results[[counter]] <- res

  # -------------------------
  # 🔥 Plot AUC across ALL MODELS (inner CV leaderboard)
  # -------------------------
  p <- ggplot(res$inner_cv_metrics,
              aes(x = reorder(model_id, AUC), y = AUC)) +
    geom_col(fill = "steelblue") +
    coord_flip() +
    ggtitle(paste("Inner CV AUC across H2O models - Fold:", fold_name)) +
    xlab("Model ID") + ylab("AUC")

  print(p)   # display plot
  plot_list[[counter]] <- p

  counter <- counter + 1
}
```
